# Comandos basicos de git

## Conocer el estado de mi work directory

'git status'

# agregar archivos o directorios a git

'git add' 

'git add directorio' directorio

'git add nombre_archivo' un archivo

'git add nombre_archivo otro_archivo' varios archivos

'git add -a' todo

'git add .' todo

## Confirmar los cambios

'git commit -n' "mensaje descriptivo"

## Conocer el historial del proyecto

'git log'

## Agregar repositorio remoto

'git remote add {nombre_remoto} {URL}'

## Verificar mis repositorios remotos

'git remote -v'

## Verificar las ramas de mi repo

'git branch'

## Renombrar una rama de mi repo

'git branch -m [NUEVO_NOMBRE]'
